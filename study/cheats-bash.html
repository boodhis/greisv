Просмотр

Пайпы и перенаправления

ps aux | grep nginx
cat access.log | awk '{print $1}' | sort | uniq -c | sort -rn | head
echo "text" > f          # перезаписать
echo "text" >> f         # дописать
cmd 2>&1 | tee out.log
cmd > out.log 2>&1
